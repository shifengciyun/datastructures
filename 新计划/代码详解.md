## 6. 工具函数

### 6.1 SafeMalloc - 安全内存分配

```c
void *SafeMalloc(size_t size)
{
    void *ptr = malloc(size);       // 申请内存
    if (!ptr) {                     // 如果失败
        fprintf(stderr, "错误: 内存分配失败\n");
        exit(EXIT_FAILURE);         // 直接退出程序
    }
    memset(ptr, 0, size);           // 清零初始化
    return ptr;
}
```

**作用**：封装 `malloc`，分配失败时直接退出，避免野指针。`memset` 清零防止使用未初始化内存。

### 6.2 SafeStrDup - 安全字符串复制

```c
char *SafeStrDup(const char *src)
{
    if (!src) return NULL;          // 空指针检查
    size_t len = strlen(src) + 1;   // 计算长度（含'\0'）
    char *dst = SafeMalloc(len);    // 分配内存
    strncpy(dst, src, len - 1);     // 复制字符串
    dst[len - 1] = '\0';            // 确保以'\0'结尾
    return dst;
}
```

**作用**：安全复制字符串，防止缓冲区溢出。

### 6.3 BKDRHash - 哈希函数

```c
unsigned int BKDRHash(const char *str)
{
    unsigned int seed = 131;        // 种子数（素数）
    unsigned int hash = 0;
    while (*str) {
        hash = hash * seed + (unsigned char)(*str++); // 逐字符计算
    }
    return hash % HASH_SIZE;        // 取模映射到桶索引
}
```

**作用**：将用户名字符串映射为哈希表索引。BKDR算法通过乘法和加法让不同字符串分布更均匀，减少冲突。

**运行示例**：
```
输入: "zhangsan"
计算: hash = ((((0*131+'z')*131+'h')*131+...) % 101
结果: 假设为 42
则存入 g_students.buckets[42]
```

### 6.4 FormatTime - 时间格式化

```c
const char *FormatTime(time_t t)
{
    static char buf[64];                    // 静态缓冲区
    struct tm *tm_info = localtime(&t);     // 转为本地时间结构
    strftime(buf, sizeof(buf), "%Y-%m-%d %H:%M", tm_info); // 格式化
    return buf;
}
```

**作用**：将 Unix 时间戳转为 "2026-06-16 14:30" 格式。`static` 保证返回指针有效。

## 7. 链表实现

### 7.1 ListCreateNode - 创建节点

```c
Node *ListCreateNode(void *data)
{
    Node *node = SafeMalloc(sizeof(Node)); // 分配节点内存
    node->data = data;                     // 设置数据指针
    node->next = NULL;                     // 下一个节点为空
    return node;
}
```

**作用**：创建一个新节点，`void *data` 可指向 Student、Seat 或 Reservation。

### 7.2 ListPushFront - 头插法

```c
void ListPushFront(Node **head, void *data)
{
    Node *node = ListCreateNode(data); // 创建新节点
    node->next = *head;                // 新节点指向原头节点
    *head = node;                      // 头指针指向新节点
}
```

**作用**：在链表头部插入新节点，时间复杂度 O(1)。

**图示**：
```
插入前: head -> [A] -> [B] -> [C] -> NULL
插入X:  head -> [X] -> [A] -> [B] -> [C] -> NULL
```

### 7.3 ListFind - 查找节点

```c
Node *ListFind(Node *head, const void *key,
               int (*cmp)(const void *, const void *))
{
    Node *cur = head;
    while (cur) {
        if (cmp(cur->data, key) == 0) return cur; // 找到则返回
        cur = cur->next;                           // 继续遍历
    }
    return NULL; // 未找到
}
```

**作用**：遍历链表查找元素。`cmp` 是比较函数指针，支持不同类型的查找。

### 7.4 ListRemove - 删除节点

```c
ErrorCode ListRemove(Node **head, const void *key,
                     int (*cmp)(const void *, const void *),
                     void (*freeData)(void *))
{
    Node *prev = NULL, *cur = *head;
    while (cur) {
        if (cmp(cur->data, key) == 0) {  // 找到目标
            if (prev) prev->next = cur->next; // 前驱跳过当前
            else *head = cur->next;           // 删除的是头节点
            if (freeData) freeData(cur->data); // 释放数据
            free(cur);                         // 释放节点
            return SUCCESS;
        }
        prev = cur;
        cur = cur->next;
    }
    return ERROR_NOT_FOUND;
}
```

**作用**：删除链表中匹配的节点，需要维护前驱指针。

**图示**（删除B）：
```
删除前: prev -> [A] -> [B] -> [C] -> NULL
删除后: [A] -> [C] -> NULL
```

### 7.5 ListDestroy - 销毁链表

```c
void ListDestroy(Node **head, void (*freeData)(void *))
{
    Node *cur = *head;
    while (cur) {
        Node *next = cur->next;        // 保存下一个节点
        if (freeData) freeData(cur->data); // 释放数据
        free(cur);                     // 释放节点
        cur = next;                    // 继续下一个
    }
    *head = NULL; // 头指针置空
}
```

**作用**：释放整个链表的所有内存，防止内存泄漏。

### 7.6 ListCount - 计算链表长度

```c
int ListCount(Node *head)
{
    int count = 0;
    while (head) { count++; head = head->next; }
    return count;
}
```

**作用**：遍历计数，返回节点数量。

## 8. 哈希表实现

### 8.1 HashInit - 初始化哈希表

```c
void HashInit(HashTable *ht)
{
    memset(ht->buckets, 0, sizeof(ht->buckets)); // 所有桶置空
    ht->count = 0;
}
```

**作用**：初始化哈希表，所有桶指针设为 NULL。

### 8.2 HashInsert - 插入元素

```c
ErrorCode HashInsert(HashTable *ht, const char *key, void *data)
{
    unsigned int idx = BKDRHash(key);  // 计算哈希索引
    Node *node = ListCreateNode(data); // 创建节点
    node->next = ht->buckets[idx];     // 头插法
    ht->buckets[idx] = node;
    ht->count++;
    return SUCCESS;
}
```

**作用**：将学生按用户名哈希后存入对应桶中。冲突时用链地址法（同一个桶用链表存储）。

**图示**：
```
buckets[0]  -> NULL
buckets[1]  -> [StudentA] -> NULL
buckets[2]  -> NULL
...
buckets[42] -> [StudentC] -> [StudentD] -> NULL  // 冲突，链表存储
...
buckets[100] -> NULL
```

### 8.3 HashFind - 查找元素

```c
void *HashFind(HashTable *ht, const char *key)
{
    unsigned int idx = BKDRHash(key);  // 计算索引
    Node *cur = ht->buckets[idx];      // 定位到桶
    while (cur) {
        Student *s = (Student *)cur->data;
        if (strcmp(s->name, key) == 0) return cur->data; // 匹配用户名
        cur = cur->next;
    }
    return NULL; // 未找到
}
```

**作用**：根据用户名快速查找学生。先哈希定位桶，再在桶内链表中查找。

**时间复杂度**：
- 平均 O(1)：哈希均匀分布时
- 最坏 O(n)：所有元素都在同一个桶时

### 8.4 HashDestroy - 销毁哈希表

```c
void HashDestroy(HashTable *ht, void (*freeData)(void *))
{
    for (int i = 0; i < HASH_SIZE; i++) {
        ListDestroy(&ht->buckets[i], freeData); // 销毁每个桶的链表
    }
    ht->count = 0;
}
```

**作用**：释放哈希表所有内存。

## 9. 队列实现

### 9.1 QueueInit - 初始化队列

```c
void QueueInit(Queue *q)
{
    q->front = q->rear = NULL;
    q->count = 0;
}
```

### 9.2 QueuePush - 入队

```c
void QueuePush(Queue *q, void *data)
{
    Node *node = ListCreateNode(data);
    if (q->rear) q->rear->next = node; // 队尾指向新节点
    else q->front = node;              // 队列为空时，队头也指向新节点
    q->rear = node;                    // 更新队尾
    q->count++;
}
```

**作用**：在队尾添加元素。时间复杂度 O(1)。

### 9.3 QueuePop - 出队

```c
void *QueuePop(Queue *q)
{
    if (!q->front) return NULL;    // 队列为空
    Node *node = q->front;         // 取队头
    void *data = node->data;       // 保存数据
    q->front = node->next;         // 队头后移
    if (!q->front) q->rear = NULL; // 队列变空，队尾也置空
    free(node);                    // 释放节点
    q->count--;
    return data;
}
```

**作用**：从队头取出元素。时间复杂度 O(1)。

**队列图示**：
```
入队A: front -> [A] <- rear
入队B: front -> [A] -> [B] <- rear
出队:  front -> [B] <- rear   (返回A)
```

### 10.2 StudentRegister - 注册

```c
ErrorCode StudentRegister(const char *name, const char *password)
{
    // 1. 参数校验
    if (!name || !password) return ERROR_NULL_PTR;
    if (strlen(name) == 0 || strlen(password) == 0)
        return ERROR_INVALID_PARAM;

    // 2. 检查用户名是否已存在
    if (HashFind(&g_students, name))
        return ERROR_ALREADY_EXISTS;

    // 3. 创建学生对象
    Student *s = SafeMalloc(sizeof(Student));
    s->id = g_nextStudentId++;           // 分配学号
    strncpy(s->name, name, MAX_NAME_LEN - 1);
    strncpy(s->password, password, MAX_PASS_LEN - 1);
    s->creditScore = INITIAL_CREDIT;     // 初始信用分100

    // 4. 存入哈希表
    HashInsert(&g_students, name, s);
    printf("注册成功！学号: %d\n", s->id);
    return SUCCESS;
}
```

**流程**：参数校验 → 重名检查 → 创建学生 → 存入哈希表

### 10.3 StudentLogin - 登录

```c
ErrorCode StudentLogin(const char *name, const char *password)
{
    if (!name || !password) return ERROR_NULL_PTR;

    // 1. 按用户名查找
    Student *s = HashFind(&g_students, name);
    if (!s) return ERROR_NOT_FOUND;

    // 2. 验证密码
    if (strcmp(s->password, password) != 0)
        return ERROR_NOT_FOUND;  // 密码错误也返回"未找到"，防止暴力破解

    // 3. 设置当前用户
    g_currentUserId = s->id;
    printf("登录成功！欢迎 %s (信用分: %d)\n", s->name, s->creditScore);
    return SUCCESS;
}
```

**安全设计**：密码错误返回 `ERROR_NOT_FOUND` 而非专门的密码错误，防止攻击者通过错误信息判断用户名是否存在。

### 10.4 StudentLogout - 退出登录

```c
void StudentLogout(void)
{
    g_currentUserId = -1;  // 重置当前用户
    printf("已退出登录\n");
}
```

### 10.5 FindStudentById - 按ID查找学生

```c
Student *FindStudentById(int id)
{
    for (int i = 0; i < HASH_SIZE; i++) {  // 遍历所有桶
        Node *cur = g_students.buckets[i];
        while (cur) {                      // 遍历桶内链表
            Student *s = (Student *)cur->data;
            if (s->id == id) return s;     // 找到
            cur = cur->next;
        }
    }
    return NULL; // 未找到
}
```

**注意**：这是按学号查找，需要遍历整个哈希表（因为哈希表是按用户名索引的）。效率 O(n)。

### 10.6 UpdateCredit - 更新信用分

```c
ErrorCode UpdateCredit(int studentId, int delta)
{
    Student *s = FindStudentById(studentId);
    if (!s) return ERROR_NOT_FOUND;
    s->creditScore += delta;        // 增加或减少
    if (s->creditScore < 0) s->creditScore = 0; // 不允许负数
    return SUCCESS;
}
```

**作用**：签退奖励 +1，超时扣分 -5，取消扣分 -2。

## 11. 座位管理模块

### 11.1 InitSeats - 初始化座位

```c
ErrorCode InitSeats(int roomId, int count)
{
    if (count <= 0 || count > MAX_SEATS) return ERROR_INVALID_PARAM;

    // 清空旧数据
    ListDestroy(&g_seats, free);
    ListDestroy(&g_reservations, free);
    g_seatCount = 0;
    g_nextResId = 1;

    // 创建新座位
    for (int i = 1; i <= count; i++) {
        Seat *seat = SafeMalloc(sizeof(Seat));
        seat->id = i;
        seat->roomId = roomId;
        seat->status = SEAT_AVAILABLE;  // 全部空闲
        seat->studentId = 0;            // 无人使用
        ListPushFront(&g_seats, seat);
        g_seatCount++;
    }
    printf("已初始化 %d 个座位 (自习室 %d)\n", count, roomId);
    return SUCCESS;
}
```

**作用**：管理员初始化自习室，创建指定数量的座位。

### 11.2 FindSeatById - 按座位号查找

```c
Seat *FindSeatById(int seatId)
{
    Node *cur = g_seats;
    while (cur) {
        Seat *seat = (Seat *)cur->data;
        if (seat->id == seatId) return seat;
        cur = cur->next;
    }
    return NULL;
```

## 12. 预约管理模块

### 12.1 IsTimeOverlap - 时间冲突检测

```c
int IsTimeOverlap(time_t s1, time_t e1, time_t s2, time_t e2)
{
    return s1 < e2 && s2 < e1;  // 两段时间重叠的数学判断
}
```

**原理**：两个时间段 [s1,e1] 和 [s2,e2] 重叠的充要条件是 `s1 < e2 && s2 < e1`。

**图示**：
```
情况1（重叠）:  s1----s2====e1----e2
情况2（不重叠）: s1----e1    s2----e2
```

### 12.2 CheckConflict - 检查预约冲突

```c
ErrorCode CheckConflict(int seatId, time_t start, time_t end)
{
    Node *cur = g_reservations;
    while (cur) {
        Reservation *r = (Reservation *)cur->data;
        // 只检查同一座位的有效预约
        if (r->seatId == seatId &&
            (r->status == RES_PENDING || r->status == RES_ACTIVE)) {
            if (IsTimeOverlap(start, end, r->startTime, r->endTime)) {
                return ERROR_TIME_CONFLICT;  // 发现冲突
            }
        }
        cur = cur->next;
    }
    return SUCCESS;  // 无冲突
}
```

**作用**：创建预约前检查该座位在指定时间段是否已被占用。

### 12.3 CreateReservation - 创建预约

```c
ErrorCode CreateReservation(int seatId, time_t start, time_t end)
{
    // 1. 检查登录状态
    if (g_currentUserId < 0) return ERROR_STATUS;

    // 2. 检查用户存在和信用分
    Student *stu = FindStudentById(g_currentUserId);
    if (!stu) return ERROR_NOT_FOUND;
    if (stu->creditScore < 10) return ERROR_CREDIT_LOW;  // 信用分太低禁止预约

    // 3. 检查座位存在和状态
    Seat *seat = FindSeatById(seatId);
    if (!seat) return ERROR_NOT_FOUND;
    if (seat->status != SEAT_AVAILABLE) return ERROR_STATUS;  // 座位不空闲

    // 4. 检查时间合法性
    if (difftime(end, start) <= 0) return ERROR_INVALID_PARAM;

    // 5. 检查冲突
    ErrorCode err = CheckConflict(seatId, start, end);
    if (err != SUCCESS) return err;

    // 6. 创建预约记录
    Reservation *r = SafeMalloc(sizeof(Reservation));
    r->id = g_nextResId++;
    r->studentId = g_currentUserId;
    r->seatId = seatId;
    r->startTime = start;
    r->endTime = end;
    r->status = RES_PENDING;

    // 7. 更新数据结构
    ListPushFront(&g_reservations, r);
    seat->status = SEAT_RESERVED;
    seat->studentId = g_currentUserId;

    printf("预约成功！预约号: %d\n", r->id);
    return SUCCESS;
}
```

**完整流程**：
```
登录检查 → 信用分检查 → 座位检查 → 时间检查 → 冲突检查 → 创建记录 → 更新状态
```

### 12.4 CancelReservation - 取消预约

```c
ErrorCode CancelReservation(int resId)
{
    // 1. 查找预约
    Reservation *r = FindReservationById(resId);
    if (!r) return ERROR_NOT_FOUND;
    if (r->studentId != g_currentUserId) return ERROR_STATUS; // 不是自己的预约
    if (r->status != RES_PENDING) return ERROR_STATUS;        // 只能取消待签到的

    // 2. 检查取消时间
    time_t now = time(NULL);
    double diff = difftime(r->startTime, now) / 60.0;

    if (diff < CANCEL_THRESHOLD_MIN) {  // 不足30分钟
        UpdateCredit(g_currentUserId, -CREDIT_PENALTY_CANCEL); // 扣2分
        printf("取消不足%d分钟，扣除%d信用分\n", ...);
    }

    // 3. 更新状态
    r->status = RES_CANCELLED;
    Seat *seat = FindSeatById(r->seatId);
    if (seat) {
        seat->status = SEAT_AVAILABLE;  // 释放座位
        seat->studentId = 0;
    }

    printf("预约 %d 已取消\n", resId);
    return SUCCESS;
}
```

**规则**：
- 提前30分钟以上取消：不扣分
- 不足30分钟取消：扣2分

### 12.5 CheckIn - 签到

```c
ErrorCode CheckIn(int resId)
{
    // 1. 查找并校验
    Reservation *r = FindReservationById(resId);
    if (!r) return ERROR_NOT_FOUND;
    if (r->studentId != g_currentUserId) return ERROR_STATUS;
    if (r->status != RES_PENDING) return ERROR_STATUS;

    // 2. 检查是否超时
    time_t now = time(NULL);
    double diff = difftime(now, r->startTime) / 60.0;

    if (diff > CHECKIN_TIMEOUT_MIN) {  // 超过15分钟
        r->status = RES_TIMEOUT;
        UpdateCredit(g_currentUserId, -CREDIT_PENALTY_LATE); // 扣5分
        Seat *seat = FindSeatById(r->seatId);
        if (seat) {
            seat->status = SEAT_AVAILABLE;  // 释放座位
            seat->studentId = 0;
        }
        printf("签到超时，预约已取消，扣除%d信用分\n", ...);
        return ERROR_STATUS;
    }

    // 3. 签到成功
    r->status = RES_ACTIVE;
    Seat *seat = FindSeatById(r->seatId);
    if (seat) seat->status = SEAT_IN_USE;

    printf("签到成功！座位 %d\n", r->seatId);
    return SUCCESS;
}
```

**签到时间窗口**：预约开始时间前后15分钟内可签到。

### 14.2 LoadData - 加载数据

```c
ErrorCode LoadData(void)
{
    FILE *fp = fopen(DATA_FILE, "r");
    if (!fp) return ERROR_FILE;  // 文件不存在则跳过

    char line[MAX_LINE_LEN];
    enum { META, STUDENTS, SEATS, RESERVATIONS } section = META;

    while (fgets(line, sizeof(line), fp)) {
        line[strcspn(line, "\r\n")] = '\0';  // 去除换行符
        if (strlen(line) == 0) continue;     // 跳过空行

        // 识别段落标题
        if (strcmp(line, "[META]") == 0) { section = META; continue; }
        if (strcmp(line, "[STUDENTS]") == 0) { section = STUDENTS; continue; }
        if (strcmp(line, "[SEATS]") == 0) { section = SEATS; continue; }
        if (strcmp(line, "[RESERVATIONS]") == 0) { section = RESERVATIONS; continue; }

        // 根据当前段落解析数据
        switch (section) {
        case META:
            sscanf(line, "nextStudentId=%d", &g_nextStudentId);
            // ...
        case STUDENTS: {
            Student *s = SafeMalloc(sizeof(Student));
            sscanf(line, "%d,%49[^,],%49[^,],%d", &s->id, s->name, s->password, &s->creditScore);
            HashInsert(&g_students, s->name, s);
            break;
        }
        // ...
        }
    }

    fclose(fp);
    return SUCCESS;
}
```

**解析逻辑**：
1. 按行读取文件
2. 遇到 `[XXX]` 切换当前解析段落
3. 根据段落格式解析数据行
4. 创建对象并存入对应数据结构
